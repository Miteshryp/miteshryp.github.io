{{- /* Renders raw verse text: escaped, with links, bold and italic. Whitespace/indentation is preserved via 
. Shared by the ```verse
   code fence and the triple-quoted passthrough block. */ -}}
{{- $t := trim (string .) "\n\r" -}}
{{- $t = htmlEscape $t -}}
{{- $t = replaceRE `\[([^\]]+)\]\((https?://[^)\s]+|mailto:[^)\s]+|/[^)\s]*|#[^)\s]+)\)` `$1` $t -}}
{{- $t = replaceRE `\*\*([^*\n]+)\*\*` `$1` $t -}}
{{- $t = replaceRE `\*([^*\n]+)\*` `$1` $t -}}

{{ $t | safeHTML }}
